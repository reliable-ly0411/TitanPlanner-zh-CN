using System;
using System.Collections.Generic;
using System.Drawing;
using System.Globalization;
using System.Linq;
using System.Windows.Forms;

partial class Fixture
{
    private const string Raw = "Pause Stream Logging";
    private const string Bilingual = "Pause Stream Logging · 暂停流式日志记录";
    private static int assertions;

    private static void Check(bool condition, string failure)
    {
        assertions++;
        if (!condition) throw new Exception(failure);
    }

    private static void CheckCatalog()
    {
        foreach (string firmware in new[] { "ArduCopter2", "ArduPlane", "ArduRover", "ArduSub", "PX4",
            "ArduTracker", "Blimp", "Plugin", "", null })
        foreach (bool chinese in new[] { false, true })
        {
            bool applies = chinese && (firmware == "ArduCopter2" || firmware == "ArduPlane" || firmware == "ArduRover");
            Check(FormatAuxiliaryOption(firmware, 164, Raw, chinese) == (applies ? Bilingual : Raw), "Wrong logging translation " + firmware);
            Check(FormatAuxiliaryOption(firmware, 163, Raw, chinese) == Raw, "Wrong logging ID must stay raw");
            Check(FormatAuxiliaryOption(firmware, 165, Raw, chinese) == Raw, "Neighboring ID must stay raw");
            Check(FormatAuxiliaryOption(firmware, 164, "Pause Stream Logging (alias)", chinese) == "Pause Stream Logging (alias)", "Unreviewed alias must stay raw");
            Check(FormatAuxiliaryOption(firmware, 164, "pause stream logging", chinese) == "pause stream logging", "Changed case must stay raw");
            Check(FormatAuxiliaryOption(firmware, 164, null, chinese) == null, "Null value changed");
        }
        Check(FormatAuxiliaryOption("ArduCopter2", 78, "RunCam Control", true) == "RunCam Control · RunCam 控制", "RunCam regressed");
        Check(FormatAuxiliaryOption("ArduRover", 300, "Scripting1", true) == "Scripting1 · 脚本功能 1", "Scripting regressed");
        Check(FormatAuxiliaryOption("ArduPlane", 0, "Do Nothing", true) == "Do Nothing · 无操作", "No-op display regressed");
    }

    private void CheckDropdown(string culture, int width)
    {
        CultureInfo.CurrentUICulture = CultureInfo.GetCultureInfo(culture);
        bool chinese = culture == "zh-CN";
        var rows = new List<KeyValuePair<int, string>>
        {
            new KeyValuePair<int, string>(0, "Do Nothing"),
            new KeyValuePair<int, string>(164, Raw)
        };
        using (var window = new Form { ClientSize = new Size(430, 150) })
        using (var combo = new ComboBox { Width = width, DropDownWidth = width, DropDownStyle = ComboBoxStyle.DropDownList,
            DisplayMember = "Value", ValueMember = "Key", Font = new Font(SystemFonts.DefaultFont.FontFamily, 12) })
        {
            window.Controls.Add(combo);
            ConfigureAuxiliaryDisplay(combo);
            window.Show();
            foreach (var firmware in new[] { "ArduCopter2", "ArduRover", "ArduPlane", "PX4", "ArduCopter2" })
            {
                _auxFunctionsFirmware = firmware;
                bool applies = chinese && firmware != "PX4";
                combo.DataSource = rows.ToList();
                combo.SelectedValue = 164;
                Application.DoEvents();
                Check(combo.DrawMode == (chinese ? DrawMode.OwnerDrawFixed : DrawMode.Normal), "Locale owner-drawing scope");
                Func<int, string> visible = index => combo.DrawMode == DrawMode.OwnerDrawFixed
                    ? DisplayAuxiliaryOption(combo, index) : combo.GetItemText(combo.Items[index]);
                Check(visible(1) == (applies ? Bilingual : Raw), "Wrong rendered label: " + culture + "/" + firmware);
                combo.Focus();
                combo.DroppedDown = true;
                Application.DoEvents();
                Check(combo.DroppedDown, "Actual dropdown did not open");
                if (chinese)
                {
                    int minimum = TextRenderer.MeasureText(applies ? Bilingual : Raw, combo.Font).Width;
                    Check(combo.DropDownWidth >= minimum, "Dropdown width clipped reviewed label");
                    Check(combo.DropDownWidth <= Screen.FromControl(combo).WorkingArea.Width, "Dropdown exceeds screen");
                }
                combo.DroppedDown = false;
                Check((int)combo.SelectedValue == 164 && combo.Text == Raw, "Selection changed raw value");
                Check(combo.Items.Cast<KeyValuePair<int, string>>().SequenceEqual(rows), "The raw list changed");
            }
        }
    }

    [STAThread]
    private static int Main()
    {
        try
        {
            CheckCatalog();
            var fixture = new Fixture();
            foreach (var culture in new[] { "zh-CN", "zh-Hans", "en-US" })
            foreach (var width in new[] { 120, 220 })
                fixture.CheckDropdown(culture, width);
            Console.WriteLine("Auxiliary logging caption / dropdown: " + assertions + " assertions passed.");
            return 0;
        }
        catch (Exception error)
        {
            Console.Error.WriteLine(error);
            return 1;
        }
    }
}
