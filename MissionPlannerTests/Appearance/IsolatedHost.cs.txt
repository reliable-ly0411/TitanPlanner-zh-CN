namespace Microsoft.VisualStudio.TestTools.UnitTesting
{
    [AttributeUsage(AttributeTargets.Class)] public class TestClassAttribute : Attribute { }
    [AttributeUsage(AttributeTargets.Method)] public class TestMethodAttribute : Attribute { }
    public static class Assert
    {
        public static void IsTrue(bool value, string message = null) { if (!value) throw new Exception(message ?? "Expected true"); }
        public static void IsFalse(bool value) { IsTrue(!value); }
        public static void IsNull(object value) { IsTrue(value == null); }
        public static void IsNotNull(object value) { IsTrue(value != null); }
        public static void AreEqual<T>(T expected, T actual, string message = null) { if (!object.Equals(expected, actual)) throw new Exception(message ?? "Expected " + expected + "; actual " + actual); }
        public static void AreNotEqual<T>(T unexpected, T actual) { IsFalse(object.Equals(unexpected, actual)); }
        public static void AreSame(object expected, object actual) { IsTrue(object.ReferenceEquals(expected, actual)); }
        public static void Fail(string message) { throw new Exception(message); }
        public static void ThrowsException<T>(Action action) where T : Exception
        { try { action(); } catch (T) { return; } throw new Exception("Expected " + typeof(T).Name); }
    }
    public static class CollectionAssert
    {
        public static void AreEqual(byte[] expected, byte[] actual)
        { Assert.IsTrue(expected.SequenceEqual(actual)); }
    }
}

internal static class AppearanceIsolatedHost
{
    private static int assertions;
    private static void Check(bool value, string message)
    { assertions++; if (!value) throw new Exception(message); }
    private static Control[] Descendants(Control root)
    { return new[] { root }.Concat(root.Controls.Cast<Control>().SelectMany(Descendants)).ToArray(); }

    private static void PersistencePhase(string output, string phase)
    {
        string directory = Path.Combine(Settings.UserDirectory, "appearance");
        if (phase == "seed")
        {
            var icon = AppearanceAssetStoreTests.IconFile(output);
            var image = AppearanceAssetStoreTests.ImageFile(output, ".png", Color.White);
            DesktopAppearance.Import(AppearanceAsset.WindowIcon, icon);
            DesktopAppearance.Import(AppearanceAsset.StartupImage, image);
            File.Delete(icon); File.Delete(image);
            File.WriteAllText(Path.Combine(output, "manifest-snapshot.xml"), File.ReadAllText(Path.Combine(directory, "settings.xml")));
        }
        else if (phase == "damage")
        {
            foreach (var path in Directory.GetFiles(directory, "*.ico")) File.WriteAllText(path, "synthetic damage");
            foreach (var path in Directory.GetFiles(directory, "*.png")) File.Delete(path);
        }
        else if (phase == "reset")
        {
            DesktopAppearance.RestoreDefault(AppearanceAsset.WindowIcon);
            DesktopAppearance.RestoreDefault(AppearanceAsset.StartupImage);
        }
        else
        {
            bool custom = phase == "verify";
            Check(custom == (DesktopAppearance.WindowIcon != null), "fresh-process icon selection wrong");
            string warning;
            using (var image = DesktopAppearance.Store.LoadStartupImage(out warning))
                Check(custom == (image != null), "fresh-process startup selection wrong");
            if (phase == "fallback")
            {
                Check(DesktopAppearance.IconWarning != null && warning != null, "fallback not reported");
                Check(File.ReadAllText(Path.Combine(directory, "settings.xml")) == File.ReadAllText(Path.Combine(output, "manifest-snapshot.xml")), "fallback changed manifest");
            }
            if (phase == "defaults") Check(DesktopAppearance.IconWarning == null && warning == null, "default still warns");
            using (var splash = new MissionPlanner.Splash())
            {
                var picture = Descendants(splash).OfType<PictureBox>().Single();
                Check(custom == (picture.Image.Width == 80 && picture.Image.Height == 40), "fresh splash image wrong");
            }
        }
        Console.WriteLine("PASS fresh-process=" + phase + " assertions=" + assertions);
    }

    [STAThread]
    private static int Main(string[] args)
    {
        try
        {
            string output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
            Settings.UserDirectory = Path.Combine(output, "synthetic-user-data");
            if (args.Length > 1) { PersistencePhase(output, args[1]); return 0; }
            var tests = new AppearanceAssetStoreTests(); int passed = 0;
            foreach (var method in tests.GetType().GetMethods().Where(m => m.GetCustomAttributes(typeof(Microsoft.VisualStudio.TestTools.UnitTesting.TestMethodAttribute), false).Length != 0))
            {
                try { method.Invoke(tests, null); passed++; Console.WriteLine("PASS " + method.Name); }
                catch (TargetInvocationException ex) { throw new Exception(method.Name, ex.InnerException); }
            }
            Application.EnableVisualStyles();
            var originalIcon = ThemeManager.FormIcon;
            string source = AppearanceAssetStoreTests.IconFile(output);
            try
            {
                using (var mainWindow = new MissionPlanner.MainV2())
                using (var form = new Form())
                {
                    mainWindow.Icon = originalIcon;
                    var handle = mainWindow.Handle;
                    form.Icon = originalIcon; form.Show();
                    Check(!Application.OpenForms.Cast<Form>().Contains(mainWindow), "fixture main unexpectedly in OpenForms");
                    DesktopAppearance.Import(AppearanceAsset.WindowIcon, source);
                    Check(object.ReferenceEquals(form.Icon, ThemeManager.FormIcon), "open form did not receive custom icon");
                    Check(object.ReferenceEquals(mainWindow.Icon, ThemeManager.FormIcon), "main window absent from OpenForms kept original icon after import");
                    Check(!object.ReferenceEquals(originalIcon, form.Icon), "custom icon ignored");
                    Check(!Application.OpenForms.Cast<Form>().Contains(mainWindow), "main unexpectedly entered OpenForms before reset");
                    DesktopAppearance.RestoreDefault(AppearanceAsset.WindowIcon);
                    Check(object.ReferenceEquals(originalIcon, mainWindow.Icon), "main window absent from OpenForms did not restore default icon");
                    Check(object.ReferenceEquals(originalIcon, form.Icon), "default icon not restored");
                    Check(!Application.OpenForms.Cast<Form>().Contains(mainWindow), "reset unexpectedly registered the main window");
                }
            }
            finally { MissionPlanner.MainV2.instance = null; }
            var refreshTests = new WindowIconRefreshTests(); int refreshPassed = 0;
            foreach (var method in refreshTests.GetType().GetMethods().Where(m => m.GetCustomAttributes(typeof(Microsoft.VisualStudio.TestTools.UnitTesting.TestMethodAttribute), false).Length != 0))
            {
                try { method.Invoke(refreshTests, null); refreshPassed++; Console.WriteLine("PASS " + method.Name); }
                catch (TargetInvocationException ex) { throw new Exception(method.Name, ex.InnerException); }
            }
            Console.WriteLine("PASS window-refresh-tests=" + refreshPassed + " native-WM_GETICON=Windows-only");
            foreach (string culture in new[] { "zh-CN", "zh-Hans", "en-US" })
            {
                Thread.CurrentThread.CurrentUICulture = System.Globalization.CultureInfo.GetCultureInfo(culture);
                using (var form = new Form())
                using (var page = new MissionPlanner.GCSViews.ConfigurationView.ConfigAppearance())
                {
                    DesktopAppearance.Import(AppearanceAsset.WindowIcon, source);
                    DesktopAppearance.Import(AppearanceAsset.StartupImage, AppearanceAssetStoreTests.ImageFile(output, ".png", Color.White));
                    form.AutoScaleMode = AutoScaleMode.None; page.Dock = DockStyle.Fill; form.Controls.Add(page); form.Show();
                    // Reconstruct after externally seeding so the selectors initialize from saved preferences.
                    using (var seeded = new MissionPlanner.GCSViews.ConfigurationView.ConfigAppearance())
                    {
                        var icon = (ComboBox)seeded.Controls.Find("WindowIconPreset", true).Single();
                        var startup = (ComboBox)seeded.Controls.Find("StartupImagePreset", true).Single();
                        Check(icon.SelectedIndex == 4 && startup.SelectedIndex == 3, "saved choices not restored");
                        icon.SelectedIndex = 0;
                        startup.SelectedIndex = 0;
                    }
                    Check(DesktopAppearance.WindowIcon == null, "default icon did not reset selection");
                    string resetWarning;
                    using (var resetImage = DesktopAppearance.Store.LoadStartupImage(out resetWarning))
                        Check(resetImage == null && resetWarning == null, "default startup did not reset selection");
                    foreach (int width in new[] { 320, 640, 1024 })
                    foreach (float scale in new[] { 1f, 1.5f, 2f })
                    {
                        form.ClientSize = new Size(width, 720);
                        page.Font = new Font(SystemFonts.DefaultFont.FontFamily, SystemFonts.DefaultFont.Size * scale);
                        Application.DoEvents();
                        foreach (var label in Descendants(page).OfType<Label>())
                        {
                            Check(label.Width <= page.ClientSize.Width, "label exceeds viewport");
                            Check(label.Height >= label.PreferredHeight, "label text is clipped");
                        }
                        var last = Descendants(page).OfType<Label>().Last();
                        page.ScrollControlIntoView(last); Application.DoEvents();
                        var position = page.PointToClient(last.PointToScreen(Point.Empty));
                        Check(position.Y >= 0 && position.Y + last.Height <= page.ClientSize.Height,
                            "last status label is not reachable by scrolling");
                        page.AutoScrollPosition = Point.Empty;
                    }
                    using (var image = new Bitmap(form.Width, form.Height))
                    { form.DrawToBitmap(image, new Rectangle(Point.Empty, image.Size)); image.Save(Path.Combine(output, "settings-" + culture + ".png")); }
                }
                using (var form = new Form())
                using (var settings = new Panel { Dock = DockStyle.Fill, AutoScroll = true })
                using (var appearanceGroup = new GroupBox { AutoSize = true, AutoSizeMode = AutoSizeMode.GrowAndShrink })
                using (var embedded = new MissionPlanner.GCSViews.ConfigurationView.ConfigAppearance(embedded: true))
                {
                    form.AutoScaleMode = AutoScaleMode.None;
                    form.ClientSize = new Size(700, 320);
                    appearanceGroup.Controls.Add(embedded);
                    settings.Controls.Add(appearanceGroup);
                    form.Controls.Add(settings);
                    form.Show();
                    var iconPicker = (Button)embedded.Controls.Find("ChooseLocalWindowIcon", true).Single();
                    var startupPicker = (Button)embedded.Controls.Find("ChooseLocalStartupImage", true).Single();
                    bool enhanced = culture == "zh-CN";
                    Check(iconPicker.Text == (enhanced ? "选择本地图标…" : "Choose local icon…"), "icon picker culture fallback");
                    Check(startupPicker.Text == (enhanced ? "选择本地启动图…" : "Choose local startup image…"), "startup picker culture fallback");
                    var iconPreset = (ComboBox)embedded.Controls.Find("WindowIconPreset", true).Single();
                    var startupPreset = (ComboBox)embedded.Controls.Find("StartupImagePreset", true).Single();
                    Check(iconPreset.Items.Count == 5 && startupPreset.Items.Count == 4, "preset count changed");
                    var manifest = Path.Combine(Settings.UserDirectory, "appearance", "settings.xml");
                    string before = File.ReadAllText(manifest);
                    iconPreset.SelectedIndex = 1;
                    Check(DesktopAppearance.Store.CurrentPreset(AppearanceAsset.WindowIcon) == "default", "missing vendor icon activated");
                    Check(File.ReadAllText(manifest) == before, "missing vendor icon persisted");
                    Check(Descendants(embedded).OfType<Label>().Any(l => l.Text.Contains(enhanced ? "尚未导入" : "has no local file")),
                        "missing vendor icon not reported");
                    Check(iconPreset.AccessibleDescription == iconPreset.GetItemText(iconPreset.SelectedItem),
                        "pending selection must expose its full display caption");
                    var details = (ToolTip)embedded.GetType().GetField("details", BindingFlags.Instance | BindingFlags.NonPublic)
                        .GetValue(embedded);
                    Check(details.GetToolTip(iconPreset) == iconPreset.AccessibleDescription,
                        "pending selection tooltip does not expose the full caption");
                    // Open the actual combo box without triggering an import or selection.
                    form.ClientSize = new Size(430, 320);
                    appearanceGroup.MaximumSize = new Size(405, 0);
                    appearanceGroup.MinimumSize = new Size(405, 0);
                    Application.DoEvents();
                    using (var big = new Font(startupPreset.Font.FontFamily, 20))
                    {
                        var priorFont = startupPreset.Font;
                        startupPreset.Font = big;
                        startupPreset.DroppedDown = true;
                        Application.DoEvents();
                        Check(startupPreset.DroppedDown, "expected an open preset list");
                        Check(startupPreset.AccessibleDescription == startupPreset.GetItemText(startupPreset.SelectedItem),
                            "full default caption unavailable at enlarged font");
                        using (var graphics = startupPreset.CreateGraphics())
                        foreach (var item in startupPreset.Items)
                            Check(startupPreset.DropDownWidth >= TextRenderer.MeasureText(graphics, startupPreset.GetItemText(item),
                                startupPreset.Font, Size.Empty, TextFormatFlags.SingleLine | TextFormatFlags.NoPrefix).Width,
                                "open preset popup cannot fit the caption");
                        startupPreset.DroppedDown = false;
                        startupPreset.Font = priorFont;
                        Application.DoEvents();
                    }
                    Check(startupPreset.SelectedIndex == 0 && File.ReadAllText(manifest) == before,
                        "opening the preset list changed the selected or persisted appearance");
                    foreach (int width in new[] { 430, 700, 1100 })
                    {
                        form.ClientSize = new Size(width, 320);
                        // ConfigPlanner.GroupColumnsPanel fixes each AutoSize group's width.
                        appearanceGroup.MinimumSize = Size.Empty;
                        appearanceGroup.MaximumSize = new Size(width - 25, 0);
                        appearanceGroup.MinimumSize = new Size(width - 25, 0);
                        appearanceGroup.Width = width - 25;
                        Application.DoEvents();
                        Check(!embedded.AutoScroll, "embedded appearance uses nested scrolling");
                        Check(embedded.AutoSize, "embedded appearance does not size to its contents");
                        Check(Descendants(embedded).OfType<PictureBox>().Count() == 2, "embedded previews are missing");
                        Check(Descendants(embedded).OfType<Button>().Count() == 2, "embedded picker count changed");
                        foreach (var names in new[] { new[] { "WindowIconPreset", "ChooseLocalWindowIcon" },
                                                      new[] { "StartupImagePreset", "ChooseLocalStartupImage" } })
                        {
                            var selector = embedded.Controls.Find(names[0], true).Single();
                            var picker = embedded.Controls.Find(names[1], true).Single();
                            Check(selector.Parent == picker.Parent && picker.Left >= selector.Right &&
                                  picker.Right <= picker.Parent.ClientSize.Width, "picker is not adjacent to its preset");
                        }
                        var iconPreview = embedded.Controls.Find("WindowIconPreview", true).Single();
                        var splashPreview = embedded.Controls.Find("StartupImagePreview", true).Single();
                        Check(iconPreview.Parent == splashPreview.Parent && iconPreview.Height <= 56 &&
                              splashPreview.Height <= 56, "thumbnails are not compact");
                        Check(embedded.Height <= 220, "appearance group exceeds compact height");
                        var last = Descendants(embedded).OfType<Label>().Last();
                        settings.AutoScrollPosition = new Point(0, settings.DisplayRectangle.Height);
                        Application.DoEvents();
                        var position = settings.PointToClient(last.PointToScreen(Point.Empty));
                        Check(position.Y >= 0 && position.Y + last.Height <= settings.ClientSize.Height,
                            "embedded status is not reachable through its settings parent: " + culture + "/" + width +
                            " outer=" + settings.DisplayRectangle + " view=" + settings.ClientSize + " group=" + appearanceGroup.Bounds +
                            " embedded=" + embedded.Bounds + " last=" + last.Bounds + " scrollpos=" + settings.AutoScrollPosition +
                            " relative=" + position);
                    }
                    Console.WriteLine("PASS embedded appearance " + culture);
                }
                foreach (var color in new[] { Color.Black, Color.White })
                {
                    DesktopAppearance.Import(AppearanceAsset.StartupImage, AppearanceAssetStoreTests.ImageFile(output, ".png", color));
                    using (var splash = new MissionPlanner.Splash())
                    {
                        splash.TopMost = false; splash.Show(); Application.DoEvents();
                        var controls = Descendants(splash);
                        var picture = controls.OfType<PictureBox>().Single();
                        Check(picture.Image.Width == 80 && picture.Image.Height == 40, "custom image not used by splash");
                        using (var image = new Bitmap(splash.Width, splash.Height))
                        { splash.DrawToBitmap(image, new Rectangle(Point.Empty, image.Size)); image.Save(Path.Combine(output, "splash-" + culture + "-" + color.Name + ".png")); }
                    }
                }
            }
            Console.WriteLine("PASS store-tests=" + passed + " ui-assertions=" + assertions + " drawingDpi=96 (font stress is not Windows DPI)");
            return 0;
        }
        catch (Exception ex) { Console.Error.WriteLine(ex); return 1; }
    }
}
