        // Only the legacy metadata lookup is replaced. Its caller and grid-update delegates are production code.
        private static class ParameterMetaDataRepository
        {
            internal static ParameterDescriptionSnapshot Fixture;
            internal static string Firmware, Key;
            internal static string GetParameterMetaData(string name, string field, string firmware)
            {
                if (name != Key || firmware != Firmware || Fixture == null) throw new InvalidDataException("旧元数据样本不匹配");
                var record = Fixture.Find(name);
                switch (field)
                {
                    case ParameterMetaDataConstants.Description: return record.English;
                    case ParameterMetaDataConstants.Range: return record.Range;
                    case ParameterMetaDataConstants.Units: return record.Units;
                    case ParameterMetaDataConstants.Values:
                        return string.Join(",", record.Values.Select(v => v.Code + ": " + v.English));
                    default: throw new InvalidDataException("未知的旧元数据字段 " + field);
                }
            }
        }

        private static void Require(bool ok, string reason)
        {
            if (!ok) throw new InvalidDataException(reason);
        }

        private DescriptionSession Session(ParameterDescriptionSnapshot snapshot)
        {
            var vehicle = MainV2.comPort.MAV;
            return new DescriptionSession
            {
                Vehicle = vehicle, Chinese = true, Context = snapshot.Context, Snapshot = snapshot,
                Identity = vehicle.cs.firmware + "|" + vehicle.VersionString + "|" + vehicle.cs.version + "|" + CultureInfo.CurrentUICulture.Name
            };
        }

        private void CheckSample(Form form, Sample s)
        {
            var vehicle = MainV2.comPort.MAV;
            vehicle.cs.firmware = s.Firmware;
            vehicle.VersionString = s.Report;
            vehicle.cs.version = s.Version;
            var context = ParameterDescriptionContext.TryCreate(s.Firmware, s.Report, s.Version);
            var known = DescriptionStore.Value.Bundled(context);
            Require(known != null && known.MatchesReportedBuild && known.SourceSha256 == s.Digest, s.Key + " 固定固件来源");
            var record = known.Find(s.Key);
            Require(record != null && record.English == s.English && record.Fingerprint == s.Fingerprint, s.Key + " 原始字段指纹");

            Params.Dock = DockStyle.Fill;
            Params.ReadOnly = true;
            Params.AllowUserToAddRows = false;
            Command.HeaderText = "Key"; Value.HeaderText = "Value"; Units.HeaderText = "Units";
            Options.HeaderText = "Options"; Desc.HeaderText = "Description";
            Params.Columns.AddRange(Command, Value, Units, Options, Desc);
            Controls.Add(Params);
            InitializeParameterDescriptions();
            var row = Params.Rows[Params.Rows.Add()];
            row.Cells[Command.Index].Value = s.Key;
            row.Cells[Value.Index].Value = s.Value;
            form.Show();
            Require(Params.IsHandleCreated && Params.Rows.Count == 1, s.Key + " WinForms 句柄和行数");
            _descriptionSession = Session(known);
            ParameterMetaDataRepository.Fixture = known;
            ParameterMetaDataRepository.Firmware = s.Firmware;
            ParameterMetaDataRepository.Key = s.Key;
            var search = new Regex(Regex.Escape(s.Keyword));
            string bilingual = s.English + "\n\n中文说明：\n" + s.Chinese;
            ReadGrid(row, s, bilingual, s.Units, true, "zh-CN");
            Require(MatchesParameterDescription(s.Key, search) && row.Cells[Command.Index].ToolTipText.Contains(s.Digest),
                    s.Key + " 已审中文检索与来源提示");

            foreach (var language in new[] { "zh-Hans", "en-US" })
            {
                CultureInfo.CurrentUICulture = CultureInfo.GetCultureInfo(language);
                // Under both legacy languages this production entry only clears the enhanced session, without fetching.
                PrepareParameterDescriptions();
                Require(_descriptionSession == null && !UseEnhancedParameterDescriptions, language + " 增强会话");
                ReadGrid(row, s, s.English, s.Units, false, language);
                Require(!MatchesParameterDescription(s.Key, search) && !row.Cells[Command.Index].ToolTipText.Contains(s.Keyword),
                        s.Key + " 旧语言残留增强中文");
            }

            if (s == Copter)
            {
                CultureInfo.CurrentUICulture = CultureInfo.GetCultureInfo("zh-CN");
                CheckWrongUnit(row, known, s, search);
            }
        }

        private void ReadGrid(DataGridViewRow row, Sample s, string expected, string units, bool enhanced, string label)
        {
            ResetParameterDescriptionRows();
            Require(startup && row.Cells[Desc.Index].Value == null && row.Cells[Command.Index].ToolTipText == "", label + " 未清空旧行");
            var rows = new List<KeyValuePair<DataGridViewRow, string>> { new KeyValuePair<DataGridViewRow, string>(row, s.Key) };
            var uiCulture = CultureInfo.CurrentUICulture;
            var session = _descriptionSession;
            Require(UseEnhancedParameterDescriptions == enhanced, label + " 语言分支");
            var worker = Task.Run(() =>
            {
                CultureInfo.CurrentUICulture = uiCulture;
                if (enhanced) LoadEnhancedMetadataAndUpdateCells(rows, s.Firmware, session);
                else LoadMetadataAndUpdateCells(rows, s.Firmware);
            });
            var wait = Stopwatch.StartNew();
            while (!worker.IsCompleted || row.Cells[Desc.Index].Value == null)
            {
                Application.DoEvents();
                if (worker.IsFaulted) worker.GetAwaiter().GetResult();
                if (wait.Elapsed.TotalSeconds > 10) throw new TimeoutException(label + " " + s.Key + " 行回填超时");
                Thread.Sleep(1);
            }
            worker.GetAwaiter().GetResult();
            string key = Convert.ToString(row.Cells[Command.Index].FormattedValue);
            string value = Convert.ToString(row.Cells[Value.Index].FormattedValue);
            string actualUnits = Convert.ToString(row.Cells[Units.Index].FormattedValue);
            string options = Convert.ToString(row.Cells[Options.Index].FormattedValue);
            string desc = Convert.ToString(row.Cells[Desc.Index].FormattedValue);
            Require(Params.ReadOnly && key == s.Key && Equals(row.Cells[Value.Index].Value, s.Value)
                && value == s.Value.ToString(CultureInfo.InvariantCulture), label + " 原键和原始整数");
            Require(actualUnits == units && options == s.Options && desc == expected, label + " " + s.Key + " 格内单位、范围、选项或说明");
            Console.WriteLine("ROW {0} | {1} | {2} | {3} | {4} | {5}", label, key, value, actualUnits,
                options.Replace("\n", "\\n"), desc.Replace("\n", "\\n"));
        }

        private void CheckWrongUnit(DataGridViewRow row, ParameterDescriptionSnapshot known, Sample s, Regex search)
        {
            using (var input = new MemoryStream(ReadDescriptionResource("ParameterDescriptions.official.xml")))
            {
                var releases = XDocument.Load(input).Root.Elements("release").Where(r => (string)r.Attribute("family") == "Copter");
                var target = releases.Single(r => (string)r.Attribute("version") == "4.6.3").Element("paramfile");
                var donor = releases.Single(r => (string)r.Attribute("version") == "4.7.1");
                target.Descendants("param").Single(p => (string)p.Attribute("name") == s.Key)
                    .ReplaceWith(new XElement(donor.Descendants("param").Single(p => (string)p.Attribute("name") == s.Key)));
                var changed = ParameterDescriptionSnapshot.Parse(Encoding.UTF8.GetBytes(target.ToString()), known.Context);
                var record = changed.Find(s.Key);
                Require(changed.MatchesReportedBuild && record.English == s.English
                    && record.Fingerprint == "74ba32412e8eb981db82fbb43f8c79c97026b425311df7c1c2d16e49db56cb28", "不同单位的官方记录");
                _descriptionSession = Session(changed);
                ReadGrid(row, s, s.English + "\n\n[中文说明待复核：英文或关联字段已变化]", "KiB", true, "zh-CN/wrong-unit");
                Require(!MatchesParameterDescription(s.Key, search) && !row.Cells[Command.Index].ToolTipText.Contains(s.Chinese),
                        "旧指纹仍命中已审译文");
            }
        }
