using System;
using System.Collections.Generic;
using System.Collections.ObjectModel;
using System.Globalization;
using System.Linq;
using System.Threading;
using System.Windows.Forms;

// Offline-only prototype. No app dependencies except WinForms.
namespace Plan29PreviewProbe
{
    public enum PlanKind { Unavailable, Mission, Fence, Rally }

    public sealed class PlanCommandCatalog
    {
        private readonly Dictionary<string, ushort> ids;
        private readonly Dictionary<string, double[]> displayMultipliers;
        private readonly HashSet<ushort> nonMission;
        private readonly HashSet<ushort> approvedOpaqueMissionIds;

        public PlanCommandCatalog(IDictionary<string, ushort> ids,
            IDictionary<string, double[]> displayMultipliers, IEnumerable<ushort> nonMission,
            IEnumerable<ushort> approvedOpaqueMissionIds = null)
        {
            this.ids = new Dictionary<string, ushort>(ids, StringComparer.Ordinal);
            this.displayMultipliers = displayMultipliers.ToDictionary(
                item => item.Key, item => (double[])item.Value.Clone(), StringComparer.Ordinal);
            this.nonMission = new HashSet<ushort>(nonMission);
            this.approvedOpaqueMissionIds = new HashSet<ushort>(approvedOpaqueMissionIds ?? Array.Empty<ushort>());
            if (this.nonMission.Count == 0) throw new ArgumentException("Non-mission command metadata required.");
        }

        public ushort Lookup(string command, object tag)
        {
            ushort id;
            if (command == "UNKNOWN")
            {
                if (!(tag is ushort)) throw new FormatException("Unidentified command.");
                id = (ushort)tag;
                // A bare row tag gives a number, not evidence of mission-list membership.
                if (!approvedOpaqueMissionIds.Contains(id)) throw new FormatException("Unclassified command.");
            }
            else if (!ids.TryGetValue(command, out id))
                throw new FormatException("Unidentified command.");

            if (command.StartsWith("FENCE_", StringComparison.Ordinal) || command == "RALLY_POINT" ||
                nonMission.Contains(id))
                throw new FormatException("Not a mission row.");
            return id;
        }

        public double Normalize(string command, int index, double display)
        {
            if (!displayMultipliers.TryGetValue(command, out var scale)) return display;
            if (scale.Length != 7 || scale[index] == 0 || !ReadOnlyPlanReader.IsFinite(scale[index]))
                throw new FormatException("Invalid field scale.");
            double result = display / scale[index];
            if (!ReadOnlyPlanReader.IsFinite(result)) throw new FormatException("Invalid normalized field.");
            return result;
        }
    }

    public sealed class PlanHome
    {
        internal PlanHome(byte frame, double latitude, double longitude, double altitudeMetres, double displayUnitsPerMetre = 1)
        {
            Frame = frame; Latitude = latitude; Longitude = longitude; AltitudeMetres = altitudeMetres;
            DisplayUnitsPerMetre = displayUnitsPerMetre;
        }
        public int Sequence => 0;
        public byte Frame { get; }
        public double Latitude { get; }
        public double Longitude { get; }
        public double AltitudeMetres { get; }
        public double DisplayUnitsPerMetre { get; }
    }

    public sealed class PlanRow
    {
        internal PlanRow(int sequence, string key, ushort id, byte frame, double[] displayed, double[] fieldValues)
        {
            Sequence = sequence; CommandKey = key; CommandId = id; Frame = frame;
            // P1..4 then latitude/X, longitude/Y, altitude/Z, matching the planner catalog order.
            DisplayValues = Array.AsReadOnly((double[])displayed.Clone());
            Values = Array.AsReadOnly((double[])fieldValues.Clone());
        }
        public int Sequence { get; }
        public string CommandKey { get; }
        public ushort CommandId { get; }
        public byte Frame { get; }
        public ReadOnlyCollection<double> DisplayValues { get; }
        public ReadOnlyCollection<double> Values { get; }
    }

    public sealed class ReadOnlyPlanSnapshot
    {
        internal ReadOnlyPlanSnapshot(PlanHome home, PlanRow[] rows, string displayCulture)
        {
            Home = home;
            Rows = Array.AsReadOnly((PlanRow[])rows.Clone());
            DisplayCulture = displayCulture;
        }
        public PlanHome Home { get; }
        public string DisplayCulture { get; }
        public ReadOnlyCollection<PlanRow> Rows { get; }
    }

    public sealed class ReadOnlyPlanReader
    {
        private readonly DataGridView grid;
        private readonly int ownerThread;
        private readonly PlanCommandCatalog catalog;

        public ReadOnlyPlanReader(DataGridView grid, PlanCommandCatalog catalog)
        {
            this.grid = grid ?? throw new ArgumentNullException(nameof(grid));
            this.catalog = catalog ?? throw new ArgumentNullException(nameof(catalog));
            ownerThread = Thread.CurrentThread.ManagedThreadId;
        }

        public ReadOnlyPlanSnapshot Capture(PlanKind kind, string homeLat, string homeLon, string homeAltitude,
            double homeDisplayUnitsPerMetre, byte homeFrame, CultureInfo culture)
        {
            if (Thread.CurrentThread.ManagedThreadId != ownerThread || grid.InvokeRequired || grid.IsDisposed)
                throw new InvalidOperationException("Snapshot must be captured on the owning planner UI thread.");
            if (culture == null) throw new ArgumentNullException(nameof(culture));
            if (kind != PlanKind.Mission) throw new InvalidOperationException("Only committed mission rows can be previewed.");
            if (grid.IsCurrentCellInEditMode || grid.IsCurrentCellDirty)
                throw new InvalidOperationException("Finish or cancel the current edit before capturing.");
            if (homeDisplayUnitsPerMetre <= 0 || !IsFinite(homeDisplayUnitsPerMetre))
                throw new FormatException("Invalid home altitude unit.");
            double lat = Parse(homeLat, culture);
            double lon = Parse(homeLon, culture);
            double alt = Parse(homeAltitude, culture) / homeDisplayUnitsPerMetre;
            if (Math.Abs(lat) > 90 || Math.Abs(lon) > 180 || !IsFinite(alt))
                throw new FormatException("Invalid planned Home.");
            var home = new PlanHome(homeFrame, lat, lon, alt, homeDisplayUnitsPerMetre);

            var rows = new List<PlanRow>();
            foreach (DataGridViewRow row in grid.Rows)
            {
                if (row.IsNewRow) continue;
                try
                {
                    var command = row.Cells["Command"].Value as string;
                    if (string.IsNullOrWhiteSpace(command)) throw new FormatException("Invalid command key.");
                    var id = catalog.Lookup(command, row.Cells["Command"].Tag);
                    var frame = byte.Parse(AsText(row.Cells["Frame"].Value, culture), NumberStyles.Integer, culture);
                    var values = new double[7];
                    var displayed = new double[7];
                    for (int i = 0; i < Columns.Length; i++)
                    {
                        displayed[i] = ParseCell(row.Cells[Columns[i]].Value, culture);
                        values[i] = catalog.Normalize(command, i, displayed[i]);
                    }
                    // Do not clamp delays or infer a location from raw coordinates/slots. Display-only values.
                    rows.Add(new PlanRow(rows.Count + 1, command, id, frame, displayed, values));
                }
                catch (Exception ex) when (ex is FormatException || ex is OverflowException || ex is ArgumentException)
                {
                    throw new FormatException("Invalid plan row " + (rows.Count + 1), ex);
                }
            }
            return new ReadOnlyPlanSnapshot(home, rows.ToArray(), culture.Name);
        }

        private static readonly string[] Columns = { "Param1", "Param2", "Param3", "Param4", "Lat", "Lon", "Alt" };

        private static double ParseCell(object value, CultureInfo culture)
        {
            if (value is string text) return Parse(text, culture);
            if (value is int || value is byte || value is ushort || value is float || value is double || value is decimal)
            {
                // Avoid ToString round-tripping typed doubles through the short general format.
                double number = Convert.ToDouble(value, culture);
                if (IsFinite(number)) return number;
            }
            throw new FormatException("Invalid numeric cell value.");
        }

        private static string AsText(object value, CultureInfo culture)
        {
            if (value is string text) return text;
            if (value is int || value is byte || value is ushort || value is float || value is double || value is decimal)
                return Convert.ToString(value, culture);
            throw new FormatException("Unrecognized numeric cell value.");
        }

        private static double Parse(string text, CultureInfo culture)
        {
            // Match Double.Parse's numeric conventions for committed planner cells.
            if (!double.TryParse(text, NumberStyles.Float | NumberStyles.AllowThousands, culture, out var value) ||
                !IsFinite(value)) throw new FormatException("Invalid numeric field.");
            return value;
        }

        internal static bool IsFinite(double value) => !double.IsNaN(value) && !double.IsInfinity(value);
    }
}
